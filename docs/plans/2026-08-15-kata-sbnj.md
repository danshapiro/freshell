type ProvenanceStage =
  | 'symbolic-ref' | 'status' | 'rev-parse' | 'remote'
  | 'merge' | 'ls-remote' | 'archive' | 'extract'

function provenanceStage(command: string, args: readonly string[]): ProvenanceStage {
  if (command === 'tar') return 'extract'
  const gitArgs = args.slice(2)
  if (gitArgs[0] === 'config' && gitArgs[2]?.endsWith('.remote')) return 'remote'
  if (gitArgs[0] === 'config' && gitArgs[2]?.endsWith('.merge')) return 'merge'
  if (gitArgs[0] === 'archive') return 'archive'
  return gitArgs[0] as ProvenanceStage
}

function provenanceRunner(failure?: {
  failureAt: ProvenanceStage
  result: Omit<CommandResult, 'signal'>
}) {
  return vi.fn<RunCommand>(async (command, args) => {
    const stage = provenanceStage(command, args)
    if (failure?.failureAt === stage) return { ...failure.result, signal: null }
    if (stage === 'symbolic-ref') return ok('feature\n')
    if (stage === 'status') return ok()
    if (stage === 'rev-parse') return ok(`${HEAD}\n`)

export type SourceSnapshot = {
  branch: string
  upstreamRef: string
  head: string
  tempRoot: string
  sourceArchive: string

}

export async function prepareSourceSnapshot(
  input: Pick<CloudRunLifecycleInput, 'repoRoot' | 'runId' | 'signal'>,
  runCommand: RunCommand,
): Promise<SourceSnapshot> {
  const git = async (args: readonly string[]) =>
    runChecked(runCommand, 'git', ['-C', input.repoRoot, ...args], { signal: input.signal })

  const branch = (await git(['symbolic-ref', '--quiet', '--short', 'HEAD'])).stdout.trim()
  if (!branch) throw new Error('Cloud source requires a symbolic branch')

  const status = (await git(['status', '--porcelain=v1', '--untracked-files=all'])).stdout
  if (status !== '') throw new Error(`Cloud source worktree is not clean:\n${status}`)

  const head = (await git(['rev-parse', '--verify', 'HEAD'])).stdout.trim()
  if (!/^[0-9a-f]{40}$/.test(head)) throw new Error('Cloud source HEAD is not a full Git SHA')

  const remote = (await git(['config', '--get', `branch.${branch}.remote`])).stdout.trim()
  if (!remote || remote === '.') throw new Error('Cloud source requires a configured non-local upstream remote')
  const mergeRef = (await git(['config', '--get', `branch.${branch}.merge`])).stdout.trim()
  if (!/^refs\/heads\/[A-Za-z0-9._/-]+$/.test(mergeRef)) {
    throw new Error('Cloud source requires one configured upstream branch ref')
  }

  const live = await git(['ls-remote', '--exit-code', remote, mergeRef])
  const rows = live.stdout.trimEnd().split('\n').filter(Boolean)
  const match = rows.length === 1
    ? rows[0].match(/^([0-9a-f]{40})\t(.+)$/)
    : null
  if (!match || match[2] !== mergeRef) throw new Error('Live upstream returned no single exact branch tip')
  if (match[1] !== head) throw new Error(`Cloud source HEAD ${head} does not equal live upstream ${match[1]}`)

  const tempRoot = await mkdtemp(join(tmpdir(), `freshell-cloud-${input.runId}-`))
  const sourceArchive = join(tempRoot, 'source.tar')
  const sourceDir = join(tempRoot, 'source')
  await mkdir(sourceDir)
  try {
    await git(['archive', '--format=tar', `--output=${sourceArchive}`, 'HEAD'])
    await runChecked(runCommand, 'tar', ['-xf', sourceArchive, '-C', sourceDir], {
      signal: input.signal,
    })
  } catch (error) {
    await rm(tempRoot, { recursive: true, force: true })
    throw error
  }
  return { branch, upstreamRef: `${remote}/${mergeRef}`, head, tempRoot, sourceArchive, sourceDir }
}
